import { notFound } from 'next/navigation';
import { requireAdmin } from '@/lib/auth/session-server';
import { getAgent } from '@/services/agent.service';
import { listRegions } from '@/services/region.service';
import { PageShell } from '@/components/page-shell';
import { AgentForm } from '@/components/agent-form';

export default async function EditAgentPage({ params }: { params: Promise<{ id: string }> }) {
  const user = await requireAdmin();
  const { id } = await params;

  let agent;
  try {
    agent = await getAgent(user, id);
  } catch {
    notFound();
  }

  const activeRegions = await listRegions({ isActive: true });

  // The form only needs { id, name }. Map to that shape explicitly so
  // the type is narrow, then append the current region if it's been
  // deactivated (otherwise the select can't show the current value).
  const regionOptions: { id: string; name: string }[] = activeRegions.map((r) => ({
    id: r.id,
    name: r.name,
  }));
  if (!regionOptions.some((r) => r.id === agent.region.id)) {
    regionOptions.push({ id: agent.region.id, name: agent.region.name });
  }

  return (
    <PageShell title={`Edit ${agent.fullName}`} subtitle="Update the agent's profile.">
      <AgentForm
        mode="edit"
        agentId={agent.id}
        regions={regionOptions}
        initialValues={{
          fullName: agent.fullName,
          email: agent.email,
          phone: agent.phone,
          referralCode: agent.referralCode,
          regionId: agent.region.id,
          type: agent.type,
          status: agent.status,
          onboardingStage: agent.onboardingStage,
          payoutMethod: agent.payoutPreference.method,
          payoutProvider: agent.payoutPreference.provider ?? '',
          payoutAccountNumber: agent.payoutPreference.accountNumber ?? '',
          payoutAccountName: agent.payoutPreference.accountName ?? '',
        }}
      />
    </PageShell>
  );
}

import { notFound } from 'next/navigation';
import { requireAdmin } from '@/lib/auth/session-server';
import { getAgent, getOnboardingHistory } from '@/services/agent.service';
import { listCommissions } from '@/services/commission.service';
import { aggregateEvents } from '@/services/performance.service';
import { listAuditRows } from '@/services/audit.service';
import type { Money } from '@/lib/money/money';
import type { Currency } from '@/lib/money/currency';
import { PageShell } from '@/components/page-shell';
import { ButtonLink } from '@/components/button';
import { AgentSummaryTiles } from '@/components/agent-summary-tiles';
import { AgentTabs, type AgentTabKey } from '@/components/agent-tabs';
import { AgentProfileTab } from '@/components/agent-profile-tab';
import { AgentPerformanceTab } from '@/components/agent-performance-tab';
import { AgentCommissionsTab } from '@/components/agent-commissions-tab';
import { AgentAuditTab } from '@/components/agent-audit-tab';
import { AgentSuspendDialog } from '@/components/agent-suspend-dialog';
import { Pagination } from '@/components/pagination';

const VALID_TABS = ['profile', 'performance', 'commissions', 'audit'] as const;
const VALID_GRANULARITIES = ['daily', 'weekly', 'monthly'] as const;
const VALID_METRICS = [
  'leadsGenerated',
  'dealsClosed',
  'propertiesReferred',
  'landlordsOnboarded',
  'tenantsOnboarded',
  'revenueAttributed',
] as const;
const AUDIT_PAGE_LIMIT = 25;

export default async function AgentDetailPage({
  params,
  searchParams,
}: {
  params: Promise<{ id: string }>;
  searchParams: Promise<Record<string, string | undefined>>;
}) {
  const user = await requireAdmin();
  const { id } = await params;
  const sp = await searchParams;

  const tab: AgentTabKey =
    sp.tab && (VALID_TABS as readonly string[]).includes(sp.tab)
      ? (sp.tab as AgentTabKey)
      : 'profile';

  let agent;
  try {
    agent = await getAgent(user, id);
  } catch {
    notFound();
  }

  // Summary tiles need commission aggregates regardless of tab.
  const [approved, pending, paid, onboarding] = await Promise.all([
    listCommissions(user, { page: 1, limit: 50, agentId: id, status: 'APPROVED' }),
    listCommissions(user, { page: 1, limit: 50, agentId: id, status: 'PENDING' }),
    listCommissions(user, { page: 1, limit: 50, agentId: id, status: 'PAID' }),
    getOnboardingHistory(user, id),
  ]);

  const lastPaid = paid.items
    .map((c) => c.paidAt)
    .filter((v): v is string => Boolean(v))
    .sort()
    .pop();

  const currency: Currency = approved.items[0]?.netAmount.currency as Currency | undefined ?? 'XAF';

  function sumNet(items: { netAmount: Money }[]): Money {
    if (items.length === 0) return { amount: 0, currency };
    const first = items[0]!.netAmount.currency;
    for (const i of items) {
      if (i.netAmount.currency !== first) {
        throw new Error(`Mixed currencies: ${first} and ${i.netAmount.currency}`);
      }
    }
    const amount = items.reduce((s, i) => s + i.netAmount.amount, 0);
    return { amount, currency: first as Currency };
  }

  return (
    <PageShell
      title={agent.fullName}
      subtitle={`${agent.type === 'AGENT' ? 'Agent' : 'Ambassador'} in ${agent.region.name}`}
      action={
        <>
          {agent.status !== 'TERMINATED' && <AgentSuspendDialog agent={agent} />}
          <ButtonLink href={`/admin/agents/${agent.id}/edit`} variant="secondary">
            Edit profile
          </ButtonLink>
        </>
      }
    >
      <AgentSummaryTiles
        owedNow={sumNet(approved.items)}
        owedNowCount={approved.total}
        beingChecked={sumNet(pending.items)}
        beingCheckedCount={pending.total}
        paidToDate={sumNet(paid.items)}
        paidToDateCount={paid.total}
        lastPaidAt={lastPaid ?? null}
        onboardingStage={agent.onboardingStage}
      />

      <AgentTabs current={tab} />

      {tab === 'profile' && <AgentProfileTab agent={agent} onboarding={onboarding} />}

      {tab === 'performance' && (
        <PerformanceTabSection
          user={user}
          agentId={id}
          granularityRaw={sp.granularity}
          metricRaw={sp.metric}
          fromRaw={sp.from}
          toRaw={sp.to}
        />
      )}

      {tab === 'commissions' && (
        <CommissionsTabSection user={user} agentId={id} sp={sp} />
      )}

      {tab === 'audit' && <AuditTabSection user={user} agentId={id} sp={sp} />}
    </PageShell>
  );
}

async function PerformanceTabSection({
  user,
  agentId,
  granularityRaw,
  metricRaw,
  fromRaw,
  toRaw,
}: {
  user: Awaited<ReturnType<typeof requireAdmin>>;
  agentId: string;
  granularityRaw?: string;
  metricRaw?: string;
  fromRaw?: string;
  toRaw?: string;
}) {
  const granularity =
    granularityRaw && (VALID_GRANULARITIES as readonly string[]).includes(granularityRaw)
      ? (granularityRaw as 'daily' | 'weekly' | 'monthly')
      : 'monthly';
  const metric =
    metricRaw && (VALID_METRICS as readonly string[]).includes(metricRaw)
      ? (metricRaw as (typeof VALID_METRICS)[number])
      : 'leadsGenerated';

  const from = fromRaw ?? defaultFrom();
  const to = toRaw ?? defaultTo();

  const result = await aggregateEvents(user, {
    agentId,
    granularity,
    from,
    to,
  });

  return (
    <AgentPerformanceTab buckets={result.buckets} granularity={granularity} metric={metric} />
  );
}

async function CommissionsTabSection({
  user,
  agentId,
  sp,
}: {
  user: Awaited<ReturnType<typeof requireAdmin>>;
  agentId: string;
  sp: Record<string, string | undefined>;
}) {
  const page = Math.max(1, Number.parseInt(sp.page ?? '1', 10) || 1);
  const status = sp.status && ['PENDING', 'VERIFIED', 'APPROVED', 'PAID'].includes(sp.status)
    ? sp.status
    : undefined;
  const period = sp.period && /^\d{4}-\d{2}$/.test(sp.period) ? sp.period : undefined;

  const result = await listCommissions(user, {
    page,
    limit: 25,
    agentId,
    status,
    period,
  });

  return (
    <AgentCommissionsTab agentId={agentId} items={result.items} total={result.total} showExport />
  );
}

async function AuditTabSection({
  user,
  agentId,
  sp,
}: {
  user: Awaited<ReturnType<typeof requireAdmin>>;
  agentId: string;
  sp: Record<string, string | undefined>;
}) {
  const page = Math.max(1, Number.parseInt(sp.page ?? '1', 10) || 1);

  const result = await listAuditRows(user, {
    entityType: 'Agent',
    entityId: agentId,
    page,
    limit: AUDIT_PAGE_LIMIT,
  });

  return (
    <AgentAuditTab
      rows={result.items}
      page={result.page}
      limit={result.limit}
      total={result.total}
    />
  );
}

function defaultFrom(): string {
  const d = new Date();
  d.setUTCDate(d.getUTCDate() - 180);
  return isoDay(d);
}

function defaultTo(): string {
  return isoDay(new Date());
}

function isoDay(d: Date): string {
  return d.toISOString().slice(0, 10);
}

import { requireAdmin } from '@/lib/auth/session-server';
import { listAgents } from '@/services/agent.service';
import { listRegions } from '@/services/region.service';
import { PageShell } from '@/components/page-shell';
import { ButtonLink } from '@/components/button';
import { TableShell } from '@/components/table-shell';

import { AgentsFilterBar } from '@/components/agents-filter-bar';

import type { AgentStatus } from '@/components/status-badge';

const PAGE_SIZES = [5, 10, 25, 30, 50] as const;

export default async function AgentsPage({
  searchParams,
}: {
  searchParams: Promise<Record<string, string | undefined>>;
}) {
  const user = await requireAdmin();
  const sp = await searchParams;

  const page = Math.max(1, Number.parseInt(sp.page ?? '1', 10) || 1);
  const limitRaw = Number.parseInt(sp.limit ?? '10', 10);
  const limit = (PAGE_SIZES as readonly number[]).includes(limitRaw) ? limitRaw : 10;

  const status =
    sp.status && ['ACTIVE', 'SUSPENDED', 'INACTIVE', 'TERMINATED'].includes(sp.status)
      ? (sp.status as AgentStatus)
      : undefined;
  const type =
    sp.type && ['AGENT', 'AMBASSADOR'].includes(sp.type)
      ? (sp.type as 'AGENT' | 'AMBASSADOR')
      : undefined;
  const regionId = sp.regionId || undefined;

  const [agentsPage, regions] = await Promise.all([
    listAgents(user, { page, limit, status, type, regionId }),
    listRegions({ isActive: true }),
  ]);

  const hasActiveFilters = Boolean(status || type || regionId);
  const isEmpty = agentsPage.items.length === 0;
  const tableState = isEmpty ? 'empty' : 'loaded';
  const emptyVariant = hasActiveFilters ? 'no-matches' : 'first-use';

  // A signature of the current filter/page state. Used as the key on
  // the table so that any change to the filters or page remounts the

import { checkIpLimit, checkFailureLimit, recordFailure } from '@/lib/ratelimit';
import { getClientIp } from '@/lib/ratelimit/request-ip';

export const dynamic = 'force-dynamic';

function tooManyRequests(retryAfterSeconds: number): NextResponse {
  return NextResponse.json(
    {
      ok: false,
      error: { code: 'CONFLICT', message: 'Too many attempts. Try again shortly.' },
    },
    { status: 429, headers: { 'retry-after': String(retryAfterSeconds) } },
  );
}

export async function POST(request: Request): Promise<NextResponse> {
  try {
    const ip = getClientIp(request);

    const ipLimit = await checkIpLimit(ip);
    if (!ipLimit.success) return tooManyRequests(ipLimit.retryAfterSeconds);

    const body = await readJsonBody(request);
    const { email, password, rememberMe: remember } = parseLoginBody(body);

    const failureLimit = await checkFailureLimit(ip, email);
    if (!failureLimit.success) return tooManyRequests(failureLimit.retryAfterSeconds);

    await connectDb();
    const admin = await Admin.findOne({ email }).lean<{
      _id: unknown;
      passwordHash?: string | null;
    }>();

    if (!admin || !admin.passwordHash) {
      await recordFailure(ip, email);
      throw new UnauthorizedError('Invalid credentials');
    }

    const valid = await verifyPassword(password, admin.passwordHash);
    if (!valid) {
      await recordFailure(ip, email);
      throw new UnauthorizedError('Invalid credentials');
    }

    const { sessionId, expiresAt } = await createSession(
      {
        type: 'ADMIN',
        id: admin._id as unknown as import('mongoose').Types.ObjectId,
      },
      { rememberMe: remember },
    );

    const response = ok({ subjectType: 'ADMIN' as const });
    response.headers.append('Set-Cookie', buildSessionCookie(sessionId, expiresAt, remember));
    return response;
  } catch (err) {
    return toErrorResponse(err, undefined);
  }
}

/**
 * Logs out the caller. Idempotent: logging out when not logged in still
 * succeeds and clears the cookie.
 */
export async function POST(request: Request): Promise<NextResponse> {
  try {
    const sessionId = readSessionCookieFromHeader(request.headers.get('cookie'));
    if (sessionId) {
      await revokeSession(sessionId);

    return response;
  } catch (err) {
    return toErrorResponse(err);
  }
}

  'id',
  'fullName',
  'type',

  'phone',
  'region',

  'onboardingStage',
  'referralCode',

import { requireSessionUser } from '@/lib/auth/session-server';

import { formatAmountOnly } from '@/lib/money/format';
import { CommissionStatusBadge } from '@/components';

import Link from 'next/link';

const STATUSES = ['PENDING', 'VERIFIED', 'APPROVED', 'PAID'] as const;
const PAGE_SIZES = [5, 10, 25, 30, 50] as const;

export default async function LedgerPage({
  searchParams,
}: {
  searchParams: Promise<Record<string, string | undefined>>;
}) {
  const user = await requireSessionUser();
  const sp = await searchParams;

  const page = Math.max(1, Number.parseInt(sp.page ?? '1', 10) || 1);
  const limitRaw = Number.parseInt(sp.limit ?? '25', 10);
  const limit = (PAGE_SIZES as readonly number[]).includes(limitRaw) ? limitRaw : 25;
  const status = STATUSES.includes(sp.status as never) ? sp.status : undefined;
  const period = sp.period && /^\d{4}-\d{2}$/.test(sp.period) ? sp.period : undefined;

  const result = await listCommissions(user, { page, limit, status, period });

  return (
    <main className="mx-auto max-w-md px-4 py-6 pb-16">
      <header className="flex items-center justify-between">
        <h1 className="text-xl font-semibold">My ledger</h1>

          Home
        </Link>
      </header>

      <form method="get" className="mt-4 flex gap-2">
        <select
          name="status"
          defaultValue={status ?? ''}
          className="flex-1 rounded border border-neutral/30 bg-background px-2 py-1 text-sm"
        >
          <option value="">All statuses</option>
          {STATUSES.map((s) => (
            <option key={s} value={s}>
              {s}
            </option>
          ))}
        </select>

          type="month"

          defaultValue={period ?? ''}

        <button type="submit" className="btn-secondary text-sm">
          Filter

      {result.items.length === 0 ? (
        <div className="card mt-6 text-sm text-neutral">No commissions match.</div>
      ) : (
        <ul className="mt-6 space-y-2">
          {result.items.map((c) => (
            <li key={c.id} className="card">
              <div className="flex items-center justify-between">
                <p className="truncate text-sm font-medium">{c.coreCustomerId}</p>
                <span className="amount text-sm">{formatAmountOnly(c.netAmount)}</span>

                <span>{c.period}</span>
                <CommissionStatusBadge status={c.status} />
              </div>
            </li>
          ))}
        </ul>
      )}

      <Pagination page={page} limit={limit} total={result.total} />
    </main>
  );
}

  'id',
  'agentId',
  'coreCustomerId',
  'period',
  'status',
  'baseAmount',
  'grossAmount',
  'platformFee',
  'netAmount',
  'currency',
  'rateApplied',
  'createdAt',
  'paidAt',

 * GET /api/reports/commissions.csv
 * Admin-only. Returns the filtered set as CSV. Same query params as
 * GET /api/commissions.
 */
export const GET = withAuth(['ADMIN'], async (request, _context, user) => {
  const url = new URL(request.url);
  const query = parseListCommissionsQuery(url);

  const rows = await loadCommissionExportRows(user, {
    agentId: query.agentId,
    status: query.status,
    period: query.period,
  });

  await auditExport(
    user,
    'commissions.csv',
    {
      agentId: query.agentId ?? null,
      status: query.status ?? null,
      period: query.period ?? null,
    },
    rows.length,
  );

  const filename = `commissions-${csvTimestampSuffix()}.csv`;
  return new NextResponse(csv, {
    status: 200,
    headers: {
      'content-type': 'text/csv; charset=utf-8',
      'content-disposition': `attachment; filename="${filename}"`,
      'cache-control': 'no-store',
    },
  });
});

'use client';

import { useEffect } from 'react';

/**
 * Route-level error boundary.
 *
 * Next.js renders this in place of a route's content when a render,
 * a server component, or a descendant client component throws.
 *
 * Tone (per docs/DESIGN_BRIEF.md §8): direct, factual, no exclamation
 * marks. Never show the raw error message — it may contain internal
 * details. Show the digest instead: it's a short opaque hash that
 * identifies this error in the server logs without leaking anything.
 */
export default function RouteError({
  error,
  reset,
}: {
  error: Error & { digest?: string };
  reset: () => void;
}) {
  useEffect(() => {
    // Fire-and-forget: send the digest to the browser console so a
    // developer inspecting the page can see it. The server has already
    // logged the full error; this is a courtesy.
    // eslint-disable-next-line no-console
    console.error('[route-error]', error.digest ?? '(no digest)');
  }, [error]);

  return (
    <main className="mx-auto max-w-md px-4 py-12">
      <div className="card">
        <h1 className="text-lg font-semibold">Something went wrong</h1>
        <p className="mt-2 text-sm text-neutral">
          The page could not be loaded. You can try again, or go back to the home page.
        </p>

        {error.digest && (
          <p className="mt-3 text-xs text-neutral">
            Reference: <code className="font-mono">{error.digest}</code>
          </p>
        )}

        <div className="mt-6 flex gap-2">
          <button type="button" onClick={reset} className="btn-primary">
            Try again
          </button>
          <a href="/" className="btn-secondary">
            Go home
          </a>
        </div>
      </div>
    </main>
  );
}

// app/page.tsx
import { CopyButton } from '@/components/copy-button';
import { requireSessionUser } from '@/lib/auth/session-server';
import { formatAmountOnly, formatMoney } from '@/lib/money/format';
import { listCommissions } from '@/services/commission.service';
import { listBatches } from '@/services/payout-batch.service';
import { LogoutButton } from '@/components/logout-button';
import { CommissionStatusBadge, BatchStatusBadge } from '@/components';

const MINIMUM_BATCH_AMOUNT_XAF = 5000;

export default async function Home() {
  const user = await requireSessionUser();

  if (user.subjectType === 'ADMIN') {
    return (
      <main className="mx-auto max-w-2xl px-4 py-8">
        <h1 className="text-2xl font-semibold">EpsiHome Tracker</h1>
        <p className="mt-2 text-neutral">
          The admin dashboard has not been built yet. The API is complete and can be used directly.
        </p>
        <p className="mt-6 text-sm text-neutral">
          Signed in as <span className="font-medium">{user.email}</span>
        </p>
        <p className="mt-3">
          <LogoutButton />
        </p>
      </main>
    );
  }

  // AGENT
  const [approved, pending, recent, batches] = await Promise.all([
    listCommissions(user, { page: 1, limit: 50, status: 'APPROVED' }),
    listCommissions(user, { page: 1, limit: 50, status: 'PENDING' }),
    listCommissions(user, { page: 1, limit: 5 }),
    listBatches(user, { page: 1, limit: 5 }),
  ]);

  const approvedTotal = approved.items.reduce((sum, c) => sum + c.netAmount.amount, 0);
  const pendingTotal = pending.items.reduce((sum, c) => sum + c.netAmount.amount, 0);
  const currency = approved.items[0]?.netAmount.currency ?? 'XAF';
  const progress =
    MINIMUM_BATCH_AMOUNT_XAF > 0
      ? Math.min(100, Math.round((approvedTotal / MINIMUM_BATCH_AMOUNT_XAF) * 100))
      : 100;
  const remaining = Math.max(0, MINIMUM_BATCH_AMOUNT_XAF - approvedTotal);

  return (
    <main className="mx-auto max-w-md px-4 py-6 pb-16">
      {/* Headline: what you are owed */}
      <section className="card">
        <p className="card-title">You are owed</p>
        <p className="mt-1 text-3xl font-semibold amount">
          {formatMoney({ amount: approvedTotal, currency })}
        </p>
        {pendingTotal > 0 && (
          <p className="mt-1 text-xs text-neutral">
            plus {formatMoney({ amount: pendingTotal, currency })} pending approval
          </p>
        )}

        {approvedTotal === 0 ? (
          <p className="mt-3 text-sm text-neutral">You have no approved earnings yet.</p>
        ) : remaining > 0 ? (
          <>
            <div
              className="mt-3 h-1.5 w-full overflow-hidden rounded-full bg-neutral-bg"
              role="progressbar"
              aria-valuenow={progress}
              aria-valuemin={0}
              aria-valuemax={100}
              aria-label="Progress toward next payout"
            >
              <div className="h-full bg-accent" style={{ width: `${progress}%` }} />
            </div>
            <p className="mt-2 text-xs text-neutral">
              {formatMoney({ amount: remaining, currency })} more until your next payout.
            </p>
          </>
        ) : (
          <p className="mt-3 text-sm text-success">Ready for payout on the next cycle.</p>
        )}
      </section>

      {/* Recent commissions */}
      <section className="mt-4">
        <h2 className="mb-2 text-sm font-medium text-neutral">Recent activity</h2>
        {recent.items.length === 0 ? (
          <div className="card text-sm text-neutral">No commissions yet.</div>
        ) : (
          <ul className="space-y-2">
            {recent.items.map((c) => (
              <li key={c.id} className="card flex items-center justify-between gap-3">
                <div className="min-w-0">
                  <p className="truncate text-sm font-medium">{c.coreCustomerId}</p>
                  <p className="text-xs text-neutral">{c.period}</p>
                </div>
                <div className="flex items-center gap-2">
                  <span className="amount text-sm">{formatAmountOnly(c.netAmount)}</span>
                  <CommissionStatusBadge status={c.status} />
                </div>
              </li>
            ))}
          </ul>
        )}
      </section>

      {/* Paid history */}
      <section className="mt-4">
        <h2 className="mb-2 text-sm font-medium text-neutral">Paid history</h2>
        {batches.items.length === 0 ? (
          <div className="card text-sm text-neutral">No payouts yet.</div>
        ) : (
          <ul className="space-y-2">
            {batches.items.map((b) => (
              <li key={b.id} className="card">
                <div className="flex items-center justify-between">
                  <p className="text-sm font-medium">{b.period}</p>
                  <span className="amount text-sm">{formatAmountOnly(b.totalAmount)}</span>
                </div>
                <div className="mt-1 flex items-center justify-between text-xs text-neutral">
                  <span>
                    {b.paidAt ? `Paid ${new Date(b.paidAt).toISOString().slice(0, 10)}` : b.status}
                  </span>
                  <BatchStatusBadge status={b.status} />
                </div>
                {b.paymentReference && (
                  <p className="mt-1 text-xs text-neutral">Ref {b.paymentReference}</p>
                )}
              </li>
            ))}
          </ul>
        )}
      </section>

      {/* Profile / logout */}
      <section className="mt-8 border-t border-neutral/15 pt-4">
        <p className="text-sm font-medium">{user.name}</p>
        <p className="mt-1 flex items-center gap-2 text-xs text-neutral">
          <span>Referral code</span>
          <code className="font-mono">{user.referralCode}</code>
          <CopyButton value={user.referralCode} />
        </p>
        <p className="mt-3">
          <LogoutButton />
        </p>
      </section>
    </main>
  );
}
